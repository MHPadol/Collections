public class GetToKnowMe {

    // Method to display the pet preference
    static void displayHeader(String animal) {
        if (animal.equalsIgnoreCase("Hamster")) {
            System.out.println("🐹 🐹 🐹");
        } else {
            System.out.println("🐱 🐱 🐱");
        }
    }

    // Method to display the profile
    static void displayProfile(String name, String nickname, String birthday,
                               String address, String favSong,
                               String motivation, String support) {

        System.out.println("Name       : " + name + " / " + nickname);
        System.out.println("Birthday   : " + birthday);
        System.out.println("Address    : " + address);
        System.out.println("Fav Song   : " + favSong);
        System.out.println("Motivation : " + motivation);
        System.out.println("Support    : " + support);
    }

    public static void main(String[] args) {

        // Variables
        String animal = "Hamster";
        String name = "Mary Hazel A. Padol";
        String nickname = "Hazel";
        String birthday = "08/10/2006";
        String address = "Brgy. Navais Mandurriao";
        String favSong = "Sinta - Rob Deniel";
        String motivation = "You don't have to be great to start, but you have to start to be great. ";
        String support = "More time for assignments and less academic pressure.";

        // Method Calls
        displayHeader(animal);
        displayProfile(name, nickname, birthday, address,
                       favSong, motivation, support);
    }
}
